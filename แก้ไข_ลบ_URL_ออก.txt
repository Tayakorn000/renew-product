                    # ตรวจสอบว่ายังอยู่หน้าต่ออายุหรือไม่
                    current_url = driver.current_url
                    if "marketplace/selling/renew_listings" in current_url:
                        # อยู่หน้าต่ออายุ ใช้ JavaScript reload (ปลอดภัยที่สุด)
                        self.update_status(f"{profile_name}: รีเฟรชหน้าเว็บ...", "blue")
                        driver.execute_script("location.reload();")
                    else:
                        # ไม่ได้อยู่หน้าต่ออายุ กลับไปหน้าต่ออายุ (ใช้ URL เฉพาะกรณีนี้)
                        self.update_status(f"{profile_name}: กลับไปหน้าต่ออายุ...", "blue")
                        driver.get(renew_page_url)
                    
                    time.sleep(random.uniform(3, 5))
                    continue

===========================================
📋 สรุปการเปลี่ยนแปลง:
===========================================

❌ ลบออก:
- ทุก try-except ที่เกี่ยวกับ refresh
- ทุกโค้ดที่เกี่ยวกับ Ctrl+R
- ทุกโค้ดที่เกี่ยวกับ maximize/minimize window
- driver.get(URL) ในกรณี Error

✅ เหลือไว้:
- location.reload() เมื่ออยู่หน้าต่ออายุ (ปลอดภัย)
- driver.get(URL) เฉพาะเมื่อไปหน้าอื่น (จำเป็น)

===========================================
🎯 ผลลัพธ์:
===========================================

1. อยู่หน้าต่ออายุ → location.reload() ⭐⭐⭐⭐⭐
2. ไปหน้าอื่น → driver.get(URL) ⭐⭐ (จำเป็น)
3. Error → ไม่ทำอะไร (ปลอดภัย)
